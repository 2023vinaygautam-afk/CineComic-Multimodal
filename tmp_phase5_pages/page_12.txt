13. src/vision/clustering.py
Average-linkage clustering with cannot-link constraints.
 """Average-linkage agglomerative clustering (cosine distance) with cannot-link constraints.
Plain sklearn clustering cannot forbid two tracks that are on screen at the same time from being merged
into one character; this small implementation can. Lance-Williams update: a blocked pair has distance inf,
and inf survives averaging, so a merged cluster stays blocked from everything either part was blocked from
      ."""
from __future__ import annotations
import numpy as np
def constrained_average_linkage(E: np.ndarray, cannot_link, threshold: float) -> np.ndarray:
    """E: (n, d) unit vectors. cannot_link: iterable of (i, j). Returns labels 0..K-1 ordered by first mem
          ber."""
    n = len(E)
    if n == 0:
        return np.zeros(0, dtype=int)
    D = 1.0 - E @ E.T
    np.fill_diagonal(D, np.inf)
    for i, j in cannot_link:
        D[i, j] = D[j, i] = np.inf
    size, labels = np.ones(n), np.arange(n)
    while n > 1:
        a, b = divmod(int(np.argmin(D)), n)
        if not D[a, b] <= threshold:      # also stops when everything left is blocked (inf)
            break
        new = (size[a] * D[a] + size[b] * D[b]) / (size[a] + size[b])
        D[a, :] = new; D[:, a] = new; D[a, a] = np.inf
        D[b, :] = np.inf; D[:, b] = np.inf
        size[a] += size[b]; labels[labels == b] = a
    remap, out = {}, np.zeros(n, dtype=int)
    for i, l in enumerate(labels):
        out[i] = remap.setdefault(l, len(remap))
    return out
