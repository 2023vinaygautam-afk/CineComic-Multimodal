def build_detection_indices(fps, n_frames, segments, extra_idx, cfg) -> list[int]:
    """Frame indices to analyse: a det_fps grid inside every dialogue window (grid-aligned so overlapping
    windows share frames), plus Phase 4's candidate frames. With no transcript: the grid over the whole vi
          deo."""
    step = max(1, int(round(fps / cfg["det_fps"])))
    if not segments:
        idx = set(range(0, n_frames, step))
    else:
        idx = set()
        for s in segments:
            lo = max(0, int(round((s["start"] - cfg["window_pad_s"]) * fps)))
            hi = min(n_frames - 1, int(round((s["end"] + cfg["window_pad_s"]) * fps)))
            idx |= set(range(-(-lo // step) * step, hi + 1, step))
    idx |= {i for i in extra_idx if 0 <= i < n_frames}
    return sorted(idx)
def iter_frames(video_path, idxs):
    """Yield (frame_idx, full original-resolution BGR frame) with one sequential decode (exact frames)."""
    want = set(idxs)
    if not want:
        return
    last, i = max(want), 0
    cap = cv2.VideoCapture(str(video_path))
    while i <= last and cap.grab():
        if i in want:
            ok, fr = cap.retrieve()
            if ok:
                yield i, fr
        i += 1
    cap.release()
def check_invariants(tracks: list[dict], shots: list[dict], eps=1e-6) -> dict:
    bounds = {s["shot_id"]: (s["start"], s["end"]) for s in shots}
    span = sum(any(not (bounds[tr["shot_id"]][0] - eps <= f["t"] <= bounds[tr["shot_id"]][1] + eps)
                   for f in tr["frames"]) for tr in tracks)
    known = [tr for tr in tracks if tr["character_id"] != UNKNOWN]
    same = sum(a["character_id"] == b["character_id"] and a["t_start"] <= b["t_end"] and b["t_start"] <= a
          ["t_end"]
               for a, b in combinations(known, 2))
    return {"tracks_spanning_shots": span, "same_time_same_character": same}
