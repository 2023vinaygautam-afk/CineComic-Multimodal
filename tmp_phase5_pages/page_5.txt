def extract_frames(video_path, plan, crop_box, frames_dir: Path, fps: float,
                   jpeg_quality=90, border_thresh=16.0) -> list[dict]:
    """One sequential pass (exact frames, no seeking). Saves cropped JPEGs and computes quality stats."""
    x0, y0, x1, y1 = crop_box
    cap = cv2.VideoCapture(str(video_path))
    last, idx, records = max(plan), 0, []
    while idx <= last:
        if not cap.grab():
            log.warning("decode ended early at frame %d (expected up to %d)", idx, last)
            break
        if idx in plan:
            ok, frame = cap.retrieve()
            if ok:
                sid, fid = plan[idx]
                crop = frame[y0:y1, x0:x1]
                path = frames_dir / f"shot_{sid:03d}" / f"{fid}.jpg"
                path.parent.mkdir(parents=True, exist_ok=True)
                cv2.imwrite(str(path), crop, [cv2.IMWRITE_JPEG_QUALITY, jpeg_quality])
                rec = {"frame_id": fid, "shot_id": sid, "idx": idx, "t": idx / fps, "path": path.as_posix(
                      )}
                rec.update(frame_stats(crop, border_thresh))
                records.append(rec)
        idx += 1
    cap.release()
    return records

 from __future__ import annotations
def boundary_prf(pred: list[float], gt: list[float], tol: float = 0.5) -> dict:
    """One-to-one greedy matching of cut times within +/- tol seconds."""
    pred, gt = sorted(pred), sorted(gt)
    used, tp = set(), 0
    for p in pred:
        cands = [(abs(p - g), j) for j, g in enumerate(gt) if j not in used and abs(p - g) <= tol]
        if cands:
            used.add(min(cands)[1]); tp += 1
    P = tp / len(pred) if pred else (1.0 if not gt else 0.0)
    R = tp / len(gt) if gt else (1.0 if not pred else 0.0)
    F = 2 * P * R / (P + R) if P + R else 0.0
    return {"boundary_precision": P, "boundary_recall": R, "boundary_f1": F, "tp": tp}
def box_iou(a, b) -> float:
    ix = max(0, min(a[2], b[2]) - max(a[0], b[0])); iy = max(0, min(a[3], b[3]) - max(a[1], b[1]))
    inter = ix * iy
    union = (a[2] - a[0]) * (a[3] - a[1]) + (b[2] - b[0]) * (b[3] - b[1]) - inter
    return inter / union if union else 0.0
