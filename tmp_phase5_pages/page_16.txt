    def evaluate(self, gt: dict | None = None) -> dict:
        td = json.loads(self.out.read_text()); fd = json.loads(self.faces_path.read_text())
        shots = json.loads(self.shots_path.read_text())["shots"]
        tracks, faces = td["tracks"], fd["faces"]
        n = max(1, len(faces))
        unk_faces = sum(tr["n_faces"] for tr in tracks if tr["character_id"] == UNKNOWN)
        m = {"n_faces": len(faces), "n_tracks": len(tracks),
             "n_characters": len({tr["character_id"] for tr in tracks} - {UNKNOWN}),
             "unknown_rate": unk_faces / n,
             "mean_faces_per_track": float(np.mean([tr["n_faces"] for tr in tracks])) if tracks else 0.0,
             **check_invariants(tracks, shots)}
        if gt:
            tid = {tr["track_id"]: tr["character_id"] for tr in tracks}
            m.update(evaluate_against_gt(gt, faces, tid, self.p5["eval"]))
        return m
    def gate(self, m: dict) -> tuple[bool, str]:
        g, fails = self.p5["gate"], []
        if m["tracks_spanning_shots"]: fails.append("a track spans a shot boundary")
        if m["same_time_same_character"]: fails.append("simultaneous tracks share a character_id")
        r = m.get("recall_medium_large_frontal")
        if r is not None and r < g["medium_large_frontal_recall"]: fails.append(f"recall(medium/large fron
              tal) {r:.2f}")
        if "id_switches" in m and m["id_switches"] > g["max_id_switches"]: fails.append(f"{m['id_switches'
              ]} ID switches")
        if "cluster_purity" in m and m["cluster_purity"] < g["cluster_purity"]: fails.append(f"purity {m['
              cluster_purity']:.2f}")
        return (not fails, "; ".join(fails) or "all gates passed")
    def execute(self, gt=None, force=False) -> bool:
        t0 = time.time()
        self.run(force=force)
        m = self.evaluate(gt)
        ok, why = self.gate(m)
        m.update(runtime_s=time.time() - t0, passed=ok, reason=why, cfg=self.cache_key())
        self.metrics_path.write_text(json.dumps(m, indent=2))
        log.info("metrics: %s", json.dumps(m))
        return ok
