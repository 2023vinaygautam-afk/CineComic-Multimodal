FPS, W, H, BAR = 25, 320, 240, 20
def _make_video(path: Path, shots=3, sec=2):
    vw = cv2.VideoWriter(str(path), cv2.VideoWriter_fourcc(*"mp4v"), FPS, (W, H))
    for k in range(shots):
        rs = np.random.RandomState(k)
        tex = cv2.resize(rs.randint(40, 255, (24, 40, 3), dtype=np.uint8), (W, H - 2 * BAR), interpolation
              =cv2.INTER_NEAREST)
        for i in range(FPS * sec):
            f = np.zeros((H, W, 3), np.uint8)
            f[BAR:H - BAR] = tex
            cv2.rectangle(f, (10 + i, BAR + 10), (40 + i, BAR + 40), (255, 255, 255), -1)  # slight motion
            vw.write(f)
    vw.release()

def test_dark_and_duplicate_filters():
    q = {"min_brightness": 8, "blur_rel_thresh": 0.4, "phash_hamming": 5}
    mk = lambda i, sharp, br, ph: dict(frame_id=i, shot_id=0, t=float(i), sharp=sharp, brightness=br, phas
          h=ph)
    fr = [mk(0, 100, 2, "0" * 16), mk(1, 100, 90, "0" * 16), mk(2, 120, 90, "0" * 15 + "1"), mk(3, 100, 90
          , "f" * 16)]
    apply_filters(fr, q)
    assert fr[0]["reject"] == "dark" and fr[1]["reject"] == "duplicate" and fr[2]["reject"] is None and fr
          [3]["reject"] is None
    assert hamming_hex("0" * 16, "f" * 16) == 64
def test_pipeline_end_to_end_and_determinism(stage):
    assert stage.execute(gt={"boundaries": [2.0, 4.0], "crop_box": [0, BAR, W, H - BAR]}, force=True)
    doc1 = json.loads(stage.out.read_text())
    m = json.loads(stage.metrics_path.read_text())
    assert doc1["crop_box"] == [0, BAR, W, H - BAR]
    assert len(doc1["shots"]) == 3 and m["boundary_f1"] == 1.0
    assert m["coverage"] == 1.0 and m["candidates_outside_shot"] == 0 and m["black_border_rate"] == 0
    shot_of = {f["frame_id"]: s for s in doc1["shots"] for f in s["frames"]}
    for sid, ids in doc1["segment_candidates"].items():  # all candidates inside the segment's shot
        assert len({shot_of[i]["shot_id"] for i in ids}) == 1
    for s in doc1["shots"]:  # saved frames have no bars
        img = cv2.imread(s["frames"][0]["path"]); assert img.shape[:2] == (H - 2 * BAR, W)
    stage.run(force=True)  # same input + config => same frame list
    doc2 = json.loads(stage.out.read_text())
    assert [f["frame_id"] for s in doc1["shots"] for f in s["frames"]] == [f["frame_id"] for s in doc2["sh
          ots"] for f in s["frames"]]
