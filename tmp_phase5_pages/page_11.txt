def mouth_centre(kps) -> list[float]:
    """InsightFace 5-point order: L-eye, R-eye, nose, L-mouth-corner, R-mouth-corner."""
    return [float((kps[3][0] + kps[4][0]) / 2), float((kps[3][1] + kps[4][1]) / 2)]
def detect_faces(analyzer, frame: np.ndarray, cfg: dict) -> list[dict]:
    """Detect, filter by size / confidence, add quality + mouth. Each result carries 'emb' (np array)."""
    H, W = frame.shape[:2]
    out = []
    for d in analyzer.analyze(frame):
        x0, y0, x1, y1 = d["box"]
        x0, y0, x1, y1 = max(0, x0), max(0, y0), min(W, x1), min(H, y1)
        if min(x1 - x0, y1 - y0) < cfg["min_face_px"] or d["det"] < cfg["min_det_score"]:
            continue
        q, frontal = face_quality([x0, y0, x1, y1], d["kps"], d["det"], cfg["ref_px"])
        out.append({"box": [int(round(x0)), int(round(y0)), int(round(x1)), int(round(y1))],
                    "kps": [[round(float(x), 1), round(float(y), 1)] for x, y in d["kps"]],
                    "mouth": [round(v, 1) for v in mouth_centre(d["kps"])],
                    "det": round(d["det"], 4), "q": round(q, 4), "frontal": round(frontal, 3),
                    "emb": d["emb"]})
    return out

from __future__ import annotations
import numpy as np
from scipy.optimize import linear_sum_assignment
from src.evaluation.shot_metrics import box_iou
def _unit(v):
    return v / (np.linalg.norm(v) + 1e-12)
def track_shot(faces: list[dict], embs: np.ndarray, cfg: dict, start_id: int = 0):
    """faces: dicts with face_idx, t, box (all from ONE shot). embs[face_idx] = unit ArcFace vector.
    Returns (tracks, next_track_id); sets face['track_id']. A track never leaves the shot it was built in,
    and one face is never assigned to two tracks (one-to-one matching), so two faces in the same frame
    always get different tracks."""
    by_t: dict[float, list[dict]] = {}
    for f in faces:
        by_t.setdefault(f["t"], []).append(f)
    tracks, active, nid = [], [], start_id
    for t in sorted(by_t):
        dets = by_t[t]
        active = [tr for tr in active if t - tr["last_t"] <= cfg["max_gap_s"]]
        matched = set()
        if active:
            S = np.zeros((len(active), len(dets))); ok = np.zeros(S.shape, bool)
            for i, tr in enumerate(active):
                for j, d in enumerate(dets):
                    iou = box_iou(tr["last_box"], d["box"])
                    cos = float(tr["emb"] @ embs[d["face_idx"]])
                    S[i, j] = cfg["iou_weight"] * iou + (1 - cfg["iou_weight"]) * cos
                    ok[i, j] = ((iou >= cfg["iou_min"] or cos >= cfg["emb_gate"])
                                and cos >= cfg["emb_veto"] and S[i, j] >= cfg["min_score"])
            rows, cols = linear_sum_assignment(np.where(ok, -S, 1e6))
