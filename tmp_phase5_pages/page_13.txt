def purity(true, pred) -> float:
    by = defaultdict(Counter)
    for t, p in zip(true, pred): by[p][t] += 1
    return sum(max(c.values()) for c in by.values()) / len(true) if true else 1.0
def bcubed(true, pred) -> dict:
    ct, cp, cj = Counter(true), Counter(pred), Counter(zip(true, pred))
    n = len(true)
    if not n: return {"bcubed_precision": 1.0, "bcubed_recall": 1.0, "bcubed_f1": 1.0}
    P = sum(cj[(t, p)] / cp[p] for t, p in zip(true, pred)) / n
    R = sum(cj[(t, p)] / ct[t] for t, p in zip(true, pred)) / n
    return {"bcubed_precision": P, "bcubed_recall": R, "bcubed_f1": 2 * P * R / (P + R) if P + R else 0.0}
def id_switches(records: list[dict]) -> tuple[int, int]:
    """records: {shot_id, gt, track_id, t}. Per (shot, true person), ordered by time:
    switches = changes of track_id between consecutive matched faces; fragmentation = distinct tracks - 1.
          """
    groups = defaultdict(list)
    for r in records: groups[(r["shot_id"], r["gt"])].append(r)
    sw = fr = 0
    for rs in groups.values():
        ids = [r["track_id"] for r in sorted(rs, key=lambda r: r["t"])]
        sw += sum(a != b for a, b in zip(ids, ids[1:]))
        fr += len(set(ids)) - 1
    return sw, fr
def evaluate_against_gt(gt: dict, faces: list[dict], track_char: dict, cfg: dict) -> dict:
    """gt = {'frames': [{'t': s, 'faces': [{'box': [x0,y0,x1,y1], 'character': str?, 'frontal': bool?}]}]}
