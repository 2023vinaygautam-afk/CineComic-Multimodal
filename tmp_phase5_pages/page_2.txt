3. src/video/black_bar.py
Black-bar (letterbox) detection; crop box in original pixels.
 """Black-bar (letterbox/pillarbox) detection. Returns crop box [x0, y0, x1, y1] in ORIGINAL pixels."""
from __future__ import annotations
import cv2
import numpy as np
def crop_from_gray_frames(grays: list[np.ndarray], thresh: float, min_keep_frac: float) -> list[int]:
    H, W = grays[0].shape
    full = [0, 0, W, H]
    # A row is "bar" only if it stays dark in EVERY sampled frame (max over frames of the row mean).
    row_max = np.max([g.mean(axis=1) for g in grays], axis=0)
    rows = np.where(row_max > thresh)[0]
    if len(rows) == 0:
        return full
    y0, y1 = int(rows[0]), int(rows[-1]) + 1
    # Column profile measured only inside the content rows, so horizontal bars don't dilute it.
    col_max = np.max([g[y0:y1].mean(axis=0) for g in grays], axis=0)
    cols = np.where(col_max > thresh)[0]
    if len(cols) == 0:
        return full
