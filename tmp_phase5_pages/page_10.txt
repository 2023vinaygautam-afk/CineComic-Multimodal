 """Face detection + ArcFace embeddings (InsightFace) and per-face quality / mouth point.
All coordinates are ORIGINAL-FRAME pixels (the full decoded frame, not the black-bar-cropped one).
"""
from __future__ import annotations
import numpy as np
class InsightFaceAnalyzer:
    """SCRFD detector + ArcFace recogniser from the InsightFace 'buffalo_l' pack.
    analyze(bgr) -> [{'box': [x0,y0,x1,y1], 'kps': 5x2, 'det': float, 'emb': 512-d L2-normalised}]
    Any object with this analyze() method can replace it (the tests use a fake one)."""
    def __init__(self, model_name="buffalo_l", det_size=(640, 640), use_gpu=False, model_root=None):
        from insightface.app import FaceAnalysis  # imported lazily so the rest of the code needs no GPU s
              tack
        providers = ["CUDAExecutionProvider", "CPUExecutionProvider"] if use_gpu else ["CPUExecutionProvid
              er"]
        kw = {"root": model_root} if model_root else {}
        self.app = FaceAnalysis(name=model_name, providers=providers,
                                allowed_modules=["detection", "recognition"], **kw)
        self.app.prepare(ctx_id=0 if use_gpu else -1, det_size=tuple(det_size))
    def analyze(self, bgr: np.ndarray) -> list[dict]:
        return [{"box": f.bbox.tolist(), "kps": f.kps.tolist(), "det": float(f.det_score),
                 "emb": f.normed_embedding.astype(np.float32)} for f in self.app.get(bgr)]
