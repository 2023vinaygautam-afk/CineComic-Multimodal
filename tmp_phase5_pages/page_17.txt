    def make_track_sheet(self, tracks: list[dict]) -> None:
        """One crop per track (its best face) labelled track / character / confidence: the 'do CHAR_01 and
        CHAR_07 look like the same person?' check from the guide."""
        if not tracks:
            return
        cs, best = self.p5["track_sheet"], {tr["track_id"]: max(tr["faces"], key=lambda f: f["q"]) for tr 
              in tracks}
        need = defaultdict(list)
        for tr in tracks:
            need[best[tr["track_id"]]["idx"]].append(tr)
        tiles = {}
        for idx, frame in iter_frames(self.video_path, list(need)):
            for tr in need[idx]:
                x0, y0, x1, y1 = best[tr["track_id"]]["box"]
                mx, my = int(0.2 * (x1 - x0)), int(0.2 * (y1 - y0))
                crop = frame[max(0, y0 - my): y1 + my, max(0, x0 - mx): x1 + mx]
                t = cv2.resize(crop, (cs["thumb"], cs["thumb"]))
                cv2.putText(t, f"{tr['character_id']}", (3, 14), cv2.FONT_HERSHEY_SIMPLEX, 0.45, (0, 255, 
                      255), 1)
                cv2.putText(t, f"t{tr['track_id']} s{tr['shot_id']} {tr['id_conf']:.2f}", (3, cs["thumb"] 
                      - 5),
                            cv2.FONT_HERSHEY_SIMPLEX, 0.35, (255, 255, 0), 1)
                tiles[tr["track_id"]] = t
        arr = [tiles[k] for k in sorted(tiles)]
        if not arr:
            return
        while len(arr) % cs["cols"]:
            arr.append(np.zeros_like(arr[0]))
        sheet = np.vstack([np.hstack(arr[i:i + cs["cols"]]) for i in range(0, len(arr), cs["cols"])])
        self.results_dir.mkdir(parents=True, exist_ok=True)
        cv2.imwrite(str(self.results_dir / f"{self.video_id}_tracks.jpg"), sheet)
