    def analyze(self, bgr):
        out = []
        for name, col in COL.items():
            mask = cv2.inRange(bgr, np.clip(np.array(col) - 5, 0, 255).astype(np.uint8),
                               np.clip(np.array(col) + 5, 0, 255).astype(np.uint8))
            ys, xs = np.where(mask > 0)
            if not len(xs): continue
            x0, x1, y0, y1 = int(xs.min()), int(xs.max()) + 1, int(ys.min()), int(ys.max()) + 1
            w, h = x1 - x0, y1 - y0

            kps = [[x0 + .3 * w, y0 + .35 * h], [x0 + .7 * w, y0 + .35 * h], [x0 + .5 * w, y0 + .55 * h],
                   [x0 + .35 * w, y0 + .75 * h], [x0 + .65 * w, y0 + .75 * h]]

def _make_video(path):
    vw = cv2.VideoWriter(str(path), cv2.VideoWriter_fourcc(*"mp4v"), FPS, (W, H))
    for i in range(150):
        f = np.zeros((H, W, 3), np.uint8)
        if i < 50: _draw(f, "A", 20 + 2 * i, 50)                     # shot 0: A moves
        elif i < 100: _draw(f, "A", 30, 60); _draw(f, "B", 250, 60)   # shot 1: A and B
        else: _draw(f, "B", 150, 80)                                  # shot 2: B
        vw.write(f)
    vw.release()
